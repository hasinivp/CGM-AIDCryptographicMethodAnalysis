\begin{frame}[t]{STPA-Sec - Overview}
    \begin{alertblock}{}
        \centering
        \footnotesize\textbf{Takeaway: } STPA-Sec is a unique security analysis method connecting security failures to patient harm.
    \end{alertblock}
    \vspace{1.4cm}
    Why STPA-Sec?
    
    \begin{itemize}
        \item Top-down security analysis method with focus on control flow/feedback loops
        \item Analyze communication subsystem \& losses associated with communication failure
        \item Safety-informed evaluation criteria
    \end{itemize}

\end{frame}

\begin{frame}[t]{STPA-Sec Analysis - Unsafe Control Actions}
    \begin{alertblock}{}
        \centering
        \footnotesize\textbf{Takeaway: } Interference in glucose reading transmission can lead to severe implications for patient health/data
    \end{alertblock}
    \vspace{0.7cm}
    \begin{block}{Control Action: Transmit glucose reading (CGM to Controller)}
        \begin{itemize}
            \item UCA1: Glucose reading is blocked from being transmitted
            \item UCA2: Glucose reading is tampered before transmission
            \item UCA3: Controller is flooded with readings
            \item UCA4: Controller is flooded with readings
        \end{itemize}
    \end{block}
    
    \begin{block}{P0 Loss + Hazard}
        \begin{itemize}
            \item Hazard: Controller calculates insulin amount based on incorrect glucose reading
            \item Loss: Potential for patient harm via hyperglycemia or hypoglycemia
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}[t]{Encryption Algorithm Comparison}
    \begin{alertblock}{}
        \centering
        \footnotesize \textbf{Takeaway: } Encryption algorithms chosen based on STPA-Sec requirements with AES-CBC as a baseline/legacy
    \end{alertblock}
    \begin{table}[htbp]
    \begin{center}
    \begin{tabularx}{\textwidth}{|X|X|X|}
    \hline
    \textbf{Algorithm} & \textbf{Represents} & \textbf{Expected Tradeoff} \\
    \hline
    \texttt{AES-CBC} & \texttt{Confidentiality Only} & \texttt{No integrity, low overhead} \\
    \hline
    \texttt{AES-CBC + HMAC} & \texttt{Legacy Authenticated Encryption} & \texttt{High overhead, improved security} \\
    \hline
    \texttt{AES-CCM} & \texttt{BLE Standard} & \texttt{Secure + Efficient} \\
    \hline
    \texttt{AES-GCM} & \texttt{Modern AES} & \texttt{Modern-day Baseline} \\
    \hline
    \texttt{ChaCha20Poly1305} & \texttt{Software-optimized Modern Protocol} & \texttt{Most Software-Efficient} \\
    \hline
    \end{tabularx}

    \end{center}
\end{table}

\begin{frame}{Discussion}
    Key Takeaways         
    \begin{itemize}
        \item Overall most optimized algorithm is AES-CCM
        \item Modern AEAD protocols provided best security + limited overhead
        \item Limited computational differences between CCM \& GCM \& ChaCha - Should use security to drive choices
        \item Key size doesn't have much influence on time (small messages) - still causes overhead in terms of key storage etc.
        
        
    \end{itemize}
\end{frame}
