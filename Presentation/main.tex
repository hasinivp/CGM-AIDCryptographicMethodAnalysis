\begin{frame}[t]{STPA-Sec Analysis - Unsafe Control Actions}
    \begin{alertblock}{}
        \centering
        \footnotesize\textbf{Takeaway: } Interference in glucose reading transmission can lead to severe implications for patient health/data
    \end{alertblock}
    \vspace{0.6cm}
    \begin{block}{Control Action: Transmit glucose reading (CGM to Controller)}
        \begin{itemize}
            \item UCA1: Glucose reading is blocked from being transmitted
            \item UCA2: Glucose reading is tampered before transmission
            \item UCA3: Controller is flooded with readings
            \item UCA4: Glucose reading is replayed
        \end{itemize}
    \end{block}
    
    \begin{block}{P0 Loss + Hazard}
        \begin{itemize}
            \item Hazard: Controller calculates insulin amount based on incorrect glucose reading
            \item Loss: Potential for patient harm via hyperglycemia or hypoglycemia
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}{Discussion}
    Considerations
    \begin{itemize}
        \item Computer hardware often have AES accelerators 
        \item CGM/AID systems without this accelarator might have slower AES implementation - ChaCha is superior choice
    \end{itemize}
    \vspace{.4cm}
    Key Takeaways         
    \begin{itemize}
        \item Overall most optimized algorithm is AES-CCM
        \item Modern AEAD protocols provided best security + limited overhead
        \item Limited computational differences between CCM \& GCM \& ChaCha - Should use security to drive choices
        \item Key size doesn't have much influence on time (small messages) - still causes overhead in terms of key storage etc.
        
        
    \end{itemize}
\end{frame}
